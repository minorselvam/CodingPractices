//Controller
FileUploadController
{
	private readonly FileUploadService _fileUploadService
	
	//Service injection
	FileUploadController(FileUploadService fileUploadService)
	{
		_fileUploadService = fileUploadService
	}
	
	[HttpPost("/Upload")]
	public IActionResult Upload(File _recivedFile)
	{
		try
		{
			if(fileUploadService.ValidateFileTypes())
			{
				var result = await fileUploadService.FileUpload(_recivedFile);
				return result;
			}
		}
		catch (Exception ex)
		{
			throw ex;
		}
		return UnsupportedFile Response;
	}
}

//Service Layer
Public class FileUploadService() : IFileUpload
{
	public bool ValidateFileTypes(File _recivedFile)
	{
		//Check the file types
		If(TypeOf(_recivedFile) === "png" || "doc")
		{
			return true;
		}
		return false;
	}
	
	public async HttpResponseMessge FileUpload(File _recivedFile)
	{
		try
		{
			HttpResponseMessge finalResult;
			//Make the recived file as chunks		
			FileChunksCollection = _recivedFile.ToChunks();
			
			//Iterate each file chunk and upload the file to Azure Blob Storage Container
			Parellel.ForEach(FileChunksCollection fileChunk)
			{
				fileChunk.ToUpload("Blob Storage Container URL");
			}
			
			var finalResult = BlobContainer.Result();
			return finalResult;
		}
		catch (Exception ex)
		{
			throw ex;
		}
		return finalResult;
	}
}

Interface IFileUpload
{
	bool ValidateFileTypes(File _recivedFile);
	Async HttpResponseMessge FileUpload();
}
